// admin-login.html script const API_BASE_URL = 'https://ravel-agency-backend.onrender.com/api'; async function handleLogin(event) { event.preventDefault(); const username = document.getElementById('username').value; const password = document.getElementById('password').value; try { const response = await fetch(`${API_BASE_URL}/admin/login`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ username, password }) }); if (!response.ok) { throw new Error('Invalid credentials'); } const data = await response.json(); localStorage.setItem('adminToken', data.token); localStorage.setItem('adminUsername', data.username); window.location.href = 'admin-dashboard.html'; } catch (error) { alert('Login failed: ' + error.message); } }
